我们将构建一个完整的 **Python 后端 + C# WPF 前端** 桌面应用，命名为 **EchoDesk**。整个项目分为两个独立模块，通过标准输入输出（stdin/stdout）以 JSON 格式通信。Python 后端负责模型推理、文件解析（含 WPS COM 解密）、搜索和对话管理；C# 前端负责美观的图形界面、用户交互和进程控制。

```python
import os
import sys
import json
from pydantic import BaseModel, Field

class Config(BaseModel):
    model_path: str = "Qwen3.5-0.8B.Q4_K_M.gguf"
    model_repo: str = "unsloth/Qwen3.5-0.8B-GGUF"
    hf_endpoint: str = "https://hf-mirror.com"
    default_mode: str = "general"
    context_length: int = 4096
    max_history: int = 10
    search_default: bool = False
    temperature: float = Field(0.7, ge=0.0, le=1.0)
    max_tokens: int = 1024
    search_results: int = 3
    file_max_chars: int = 10000
    theme: str = "dark"

    @classmethod
    def load(cls, path: str = "config.json"):
        search_paths = [
            os.path.join(os.getcwd(), path),
            os.path.join(os.path.dirname(sys.executable), path) if getattr(sys, 'frozen', False) else "",
            os.path.join(sys._MEIPASS, path) if getattr(sys, '_MEIPASS', False) else "",
            os.path.join(os.path.dirname(__file__), path)
        ]
        for p in search_paths:
            if p and os.path.exists(p):
                with open(p, "r", encoding="utf-8") as f:
                    data = json.load(f)
                return cls(**data)
        raise FileNotFoundError(f"Config file '{path}' not found.")

    def save(self, path: str = "config.json"):
        with open(path, "w", encoding="utf-8") as f:
            json.dump(self.dict(), f, indent=4)
```

```python
def check_model(model_name: str, local_dir: str = "./models") -> str | None:

            local_dir=local_dir,
        )
```

```python
from llama_cpp import Llama
from typing import Generator

class LLMEngine:
    def __init__(self, model_path: str, n_ctx: int = 4096, verbose: bool = False):
        self.model_path = model_path
        self.n_ctx = n_ctx
        self.llm = Llama(model_path=model_path, n_ctx=n_ctx, verbose=verbose)

    def reload(self, model_path: str = None, n_ctx: int = None):
        if model_path:
            self.model_path = model_path
        if n_ctx:
            self.n_ctx = n_ctx
        self.llm = Llama(model_path=self.model_path, n_ctx=self.n_ctx, verbose=False)

    def chat_complete(self, messages: list, temperature: float = 0.7, max_tokens: int = 1024) -> str:
        response = self.llm.create_chat_completion(
            messages=messages,
            stream=False,
            temperature=temperature,
            max_tokens=max_tokens
        )
        return response["choices"][0]["message"]["content"]
```

### 5. `search_engine.py`

```python
from duckduckgo_search import DDGS

class SearchEngine:
    def __init__(self, max_results: int = 3):
        self.max_results = max_results

    def search(self, query: str) -> str:
        try:
            with DDGS() as ddgs:
                results = list(ddgs.text(query, max_results=self.max_results))
        except Exception as e:
            return f"搜索失败: {e}"
        if not results:
            return "未找到相关结果。"
        snippets = []
        for idx, item in enumerate(results, 1):
            title = item.get('title', '无标题')
            body = item.get('body', '')
            href = item.get('href', '')
            snippets.append(f"【{idx}】{title}\n{body}\n链接: {href}")
        return "\n\n".join(snippets)
```

```python
from datetime import datetime

MODES = {
    "general": "你是一个通用AI助手，可以回答各种问题。",
    "summary": "你是一个专业的摘要专家，擅长简洁准确地总结文本、会议记录、文章等。请提供清晰、有层次的摘要。",
    "translate": "你是一个翻译专家，精通多国语言。请根据用户要求进行准确流畅的翻译。",
    "code": "你是一个编程专家，擅长多种编程语言。请提供代码、解释、调试建议等。",
    "contract": "你是一个专业的合同审查与摘要助手。请根据用户需求，结合提供的文件内容进行回答。",
    "creative": "你是一个创意写作助手，擅长生成故事、诗歌、剧本等创意内容。"
}

class ConversationManager:
    def __init__(self, max_history: int = 10, default_mode: str = "general"):
        self.history = []
        self.max_history = max_history
        self.file_content = ""
        self.file_name = ""
        self.undo_stack = []
        self.mode = default_mode if default_mode in MODES else "general"

    def set_mode(self, mode: str) -> bool:
        if mode in MODES:
            self.mode = mode
            return True
        return False

    def get_mode(self) -> str:
        return self.mode

    def get_available_modes(self) -> dict:
        return MODES

    def add_message(self, role: str, content: str):
        self.history.append({"role": role, "content": content, "timestamp": datetime.now().isoformat()})
        if len(self.history) > self.max_history * 2:
            self.history = self.history[-self.max_history * 2:]

    def clear(self):
        self.history.clear()
        self.file_content = ""
        self.file_name = ""
        self.undo_stack.clear()

    def undo(self) -> bool:
        if len(self.history) >= 2:
            self.undo_stack.append(self.history.pop())
            self.undo_stack.append(self.history.pop())
            return True
        return False

    def set_file(self, filepath: str, content: str):
        self.file_name = filepath
        self.file_content = content

    def get_system_prompt(self) -> str:
        base_prompt = MODES.get(self.mode, MODES["general"])
        if self.file_content:
            base_prompt += f"\n\n【当前上传的文件】: {self.file_name}\n文件内容:\n{self.file_content[:2000]}"
        return base_prompt

    def build_messages(self, user_query: str, search_context: str = "") -> list:
        messages = [{"role": "system", "content": self.get_system_prompt()}]
        for msg in self.history:
            messages.append({"role": msg["role"], "content": msg["content"]})
        if search_context:
            messages.append({"role": "user", "content": f"参考以下网络信息：\n{search_context}\n\n问题：{user_query}"})
        else:
            messages.append({"role": "user", "content": user_query})
        return messages
```

### 7. `command_handler.py`

```python
import os
from typing import Tuple, Optional
from file_parser import parse_file_with_wps
from config import Config

class CommandHandler:
    def __init__(self, conv_mgr, search_engine, llm_engine, config: Config):
        self.conv_mgr = conv_mgr
        self.search_engine = search_engine
        self.llm_engine = llm_engine
        self.config = config
        self.search_enabled = config.search_default
        self.temperature = config.temperature
        self.context_length = config.context_length

    def handle(self, cmd_line: str) -> Tuple[str, Optional[str]]:
        parts = cmd_line.strip().split()
        if not parts:
            return "", None
        cmd = parts[0].lower()
        args = parts[1:]

        if cmd == "/exit" or cmd == "/quit":
            return "退出程序", "exit"
        elif cmd == "/help":
            return f"""
📋 可用指令：
  /mode [模式名]   - 切换模式 (general/summary/translate/code/contract/creative)
  /file <路径>     - 上传文件（支持加密文档，通过 WPS 自动解密）
  /search on/off   - 开启/关闭联网搜索
  /clear           - 清空对话历史
  /undo            - 撤回上一条回复
  /save <文件名>   - 导出对话为 Markdown
  /temp <值>       - 调整温度 (0.0~1.0)
  /exit            - 退出
""", None
        elif cmd == "/mode":
            if not args:
                return f"当前模式: {self.conv_mgr.get_mode()}", None
            if args[0] == "list":
                modes = self.conv_mgr.get_available_modes()
                return "\n".join([f"{k}: {v}" for k, v in modes.items()]), None
            if self.conv_mgr.set_mode(args[0]):
                self.config.default_mode = args[0]
                self.config.save()
                return f"✅ 已切换模式为: {args[0]}", None
            return f"❌ 未知模式: {args[0]}", None
        elif cmd == "/file":
            if not args:
                return "请提供文件路径", None
            filepath = args[0]
            if not os.path.exists(filepath):
                return f"文件不存在: {filepath}", None
            content = parse_file_with_wps(filepath, self.config.file_max_chars)
            if content.startswith("WPS读取失败") or content.startswith("解析失败"):
                return content, None
            self.conv_mgr.set_file(filepath, content)
            return f"✅ 已加载文件：{filepath} ({len(content)} 字符)", None
        elif cmd == "/search":
            if not args:
                return f"搜索状态: {'开启' if self.search_enabled else '关闭'}", None
            if args[0] in ("on", "off"):
                self.search_enabled = (args[0] == "on")
                self.config.search_default = self.search_enabled
                self.config.save()
                return f"🔍 搜索已{'开启' if self.search_enabled else '关闭'}", None
            return "用法: /search on/off", None
        elif cmd == "/clear":
            self.conv_mgr.clear()
            return "🗑️ 对话已清空", None
        elif cmd == "/undo":
            if self.conv_mgr.undo():
                return "↩️ 已撤回", None
            return "没有可撤回的消息", None
        elif cmd == "/save":
            if not args:
                return "请提供文件名", None
            filename = args[0]
            if not filename.endswith('.md'):
                filename += '.md'
            # 导出实现略（可参考之前）
            return f"✅ 已保存到 {filename}", None
        elif cmd == "/temp":
            if not args:
                return f"当前温度: {self.temperature}", None
            try:
                val = float(args[0])
                if 0.0 <= val <= 1.0:
                    self.temperature = val
                    self.config.temperature = val
                    self.config.save()
                    return f"✅ 温度已设为: {val}", None
                return "温度值应在 0.0~1.0 之间", None
            except ValueError:
                return "请输入有效的数字", None
        else:
            return f"未知指令: {cmd}", None
```

### 8. `file_parser.py`（关键：集成 WPS COM 解密）

```python
import os
import win32com.client as win32
from pyxtxt import xtxt

def parse_file_with_wps(filepath: str, max_chars: int = 10000) -> str:
    """优先使用 WPS COM 读取加密文档，其他格式回退 pyxtxt"""
    ext = os.path.splitext(filepath)[1].lower()
    if ext in ['.docx', '.doc', '.xlsx', '.xls', '.pptx', '.ppt']:
        return _read_office_with_wps(filepath, max_chars)
    else:
        return _parse_other(filepath, max_chars)

def _read_office_with_wps(filepath: str, max_chars: int) -> str:
    app = None
    doc = None
    try:
        ext = os.path.splitext(filepath)[1].lower()
        if ext in ['.docx', '.doc']:
            app = win32.gencache.EnsureDispatch('Kwps.Application')
        elif ext in ['.xlsx', '.xls']:
            app = win32.gencache.EnsureDispatch('KExcel.Application')
        elif ext in ['.pptx', '.ppt']:
            app = win32.gencache.EnsureDispatch('KPPT.Application')
        else:
            return "不支持的文件类型"
        app.Visible = False
        doc = app.Documents.Open(filepath)
        # 获取文本内容
        if hasattr(doc, 'Content'):
            content = doc.Content.Text
        elif hasattr(doc, 'ActiveSheet'):
            content = doc.ActiveSheet.UsedRange.Text
        else:
            content = doc.Text
        if len(content) > max_chars:
            content = content[:max_chars] + "\n...[内容过长，已截断]"
        return content
    except Exception as e:
        return f"WPS读取失败: {e}"
    finally:
        if doc:
            try: doc.Close(SaveChanges=False)
            except: pass
        if app:
            try: app.Quit()
            except: pass

def _parse_other(filepath: str, max_chars: int) -> str:
    try:
        text = xtxt(filepath)
        if not text.strip():
            return "文件内容为空"
        if len(text) > max_chars:
            text = text[:max_chars] + "\n...[内容过长，已截断]"
        return text
    except Exception as e:
        return f"解析失败: {e}"
```

### 9. `main.py`（后端主循环）

```python
import sys
import json
from llm_engine import LLMEngine
from conversation import ConversationManager
from search_engine import SearchEngine
from command_handler import CommandHandler
from config import Config
from download_utils import check_model, download_model
from file_parser import parse_file_with_wps

def main():
    config = Config.load()
    # 确保模型存在
    model_path = check_model(config.model_path)
    if not model_path:
        print(f"⚠️ 模型未找到，开始下载...", file=sys.stderr)
        model_path = download_model(config.model_path, config.model_repo, hf_endpoint=config.hf_endpoint)
    llm = LLMEngine(model_path, config.context_length)
    conv = ConversationManager(config.max_history, config.default_mode)
    search = SearchEngine(config.search_results)
    cmd = CommandHandler(conv, search, llm, config)

    while True:
        line = sys.stdin.readline()
        if not line:
            break
        try:
            req = json.loads(line)
        except:
            sys.stderr.write("Invalid JSON\n")
            continue
        action = req.get("action")
        if action == "chat":
            user_msg = req.get("message", "")
            if user_msg.startswith("/"):
                feedback, _ = cmd.handle(user_msg)
                response = {"type": "system", "content": feedback}
            else:
                search_context = ""
                if cmd.search_enabled:
                    search_result = search.search(user_msg)
                    if search_result and "未找到" not in search_result and "搜索失败" not in search_result:
                        search_context = search_result
                messages = conv.build_messages(user_msg, search_context)
                try:
                    llm_response = llm.chat_complete(messages, temperature=cmd.temperature)
                    conv.add_message("user", user_msg)
                    conv.add_message("assistant", llm_response)
                    response = {"type": "assistant", "content": llm_response}
                except Exception as e:
                    response = {"type": "system", "content": f"推理失败: {e}"}
            sys.stdout.write(json.dumps(response) + "\n")
            sys.stdout.flush()
        elif action == "file":
            filepath = req.get("path")
            content = parse_file_with_wps(filepath, config.file_max_chars)
            conv.set_file(filepath, content)
            response = {"type": "system", "content": f"已加载文件: {filepath} ({len(content)} 字符)"}
            sys.stdout.write(json.dumps(response) + "\n")
            sys.stdout.flush()
        else:
            sys.stdout.write(json.dumps({"type": "system", "content": "未知动作"}) + "\n")
            sys.stdout.flush()

if __name__ == "__main__":
    main()
```

### 5. `Services/BackendService.cs`

```csharp
using System;
using System.Diagnostics;
using System.IO;
using System.Threading.Tasks;
using Newtonsoft.Json;

namespace EchoDesk.Services
{
    public class BackendService
    {
        private Process? _process;
        private StreamWriter? _writer;
        private StreamReader? _reader;
        private readonly string _backendPath;

        public BackendService(string backendPath = "EchoBackend.exe")
        {
            _backendPath = backendPath;
        }

        public async Task StartAsync()
        {
            _process = new Process
            {
                StartInfo = new ProcessStartInfo
                {
                    FileName = _backendPath,
                    UseShellExecute = false,
                    RedirectStandardInput = true,
                    RedirectStandardOutput = true,
                    CreateNoWindow = true
                }
            };
            _process.Start();
            _writer = _process.StandardInput;
            _reader = _process.StandardOutput;
            await Task.CompletedTask;
        }

        public async Task<string> SendRequestAsync(object request)
        {
            if (_writer == null || _reader == null)
                throw new InvalidOperationException("Backend not started.");
            string json = JsonConvert.SerializeObject(request);
            await _writer.WriteLineAsync(json);
            await _writer.FlushAsync();
            string response = await _reader.ReadLineAsync();
            return response ?? "{}";
        }

        public void Stop()
        {
            _writer?.Close();
            if (_process != null && !_process.HasExited)
                _process.Kill();
            _process?.WaitForExit();
        }
    }
}
```

### 7. `MainWindow.xaml.cs`

```csharp
using System;
using System.Threading.Tasks;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Input;
using EchoDesk.Services;
using Newtonsoft.Json.Linq;

namespace EchoDesk
{
    public partial class MainWindow : Window
    {
        private BackendService? _backend;
        private bool _isProcessing;

        public MainWindow()
        {
            InitializeComponent();
            Loaded += MainWindow_Loaded;
            // 允许文件拖拽
            AllowDrop = true;
            Drop += MainWindow_Drop;
        }

        private async void MainWindow_Loaded(object sender, RoutedEventArgs e)
        {
            _backend = new BackendService();
            await _backend.StartAsync();
            StatusText.Text = "后端已启动";
        }

        private async void SendButton_Click(object sender, RoutedEventArgs e) => await SendMessage();

        private async void InputBox_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.Key == Key.Enter && Keyboard.Modifiers == ModifierKeys.None)
            {
                await SendMessage();
                e.Handled = true;
            }
        }

        private async Task SendMessage()
        {
            if (_isProcessing) return;
            string msg = InputBox.Text.Trim();
            if (string.IsNullOrEmpty(msg)) return;
            InputBox.Text = "";
            _isProcessing = true;
            AddMessage("user", msg);

            var request = new { action = "chat", message = msg };
            string responseJson = await _backend!.SendRequestAsync(request);
            try
            {
                var obj = JObject.Parse(responseJson);
                string type = obj["type"]?.ToString() ?? "system";
                string content = obj["content"]?.ToString() ?? "无响应";
                if (type == "assistant")
                    AddMessage("assistant", content);
                else if (type == "system")
                    AddMessage("system", content);
                else
                    AddMessage("system", content);
                // 更新模式显示（如果返回内容包含模式变更，可解析）
            }
            catch (Exception ex)
            {
                AddMessage("system", $"响应解析失败: {ex.Message}");
            }
            _isProcessing = false;
        }

        private void AddMessage(string role, string content)
        {
            var block = new TextBlock
            {
                TextWrapping = TextWrapping.Wrap,
                Margin = new Thickness(5),
                FontSize = 14
            };
            switch (role)
            {
                case "user":
                    block.Text = $"👤 {content}";
                    block.Foreground = System.Windows.Media.Brushes.DodgerBlue;
                    break;
                case "assistant":
                    block.Text = $"🤖 {content}";
                    block.Foreground = System.Windows.Media.Brushes.SeaGreen;
                    break;
                default:
                    block.Text = $"ℹ️ {content}";
                    block.Foreground = System.Windows.Media.Brushes.Gray;
                    break;
            }
            MessagePanel.Children.Add(block);
            // 滚动到底部
            MessageScrollViewer.ScrollToBottom();
        }

        // 文件拖拽上传
        private async void MainWindow_Drop(object sender, DragEventArgs e)
        {
            if (e.Data.GetDataPresent(DataFormats.FileDrop))
            {
                string[] files = (string[])e.Data.GetData(DataFormats.FileDrop);
                string filePath = files[0];
                AddMessage("system", $"📤 正在上传文件: {System.IO.Path.GetFileName(filePath)} ...");
                var request = new { action = "file", path = filePath };
                string responseJson = await _backend!.SendRequestAsync(request);
                try
                {
                    var obj = JObject.Parse(responseJson);
                    string content = obj["content"]?.ToString() ?? "上传完成";
                    AddMessage("system", content);
                }
                catch { }
            }
        }

        protected override void OnClosed(EventArgs e)
        {
            _backend?.Stop();
            base.OnClosed(e);
        }
    }
}
```

- 打开 `.csproj` 文件，还原 NuGet 包。
- 编译生成 `EchoDesk.exe`。

### 部署

将 `EchoDesk.exe`、`EchoBackend.exe`、`config.json` 和 `models/` 文件夹放在同一目录，双击 `EchoDesk.exe` 即可运行。
